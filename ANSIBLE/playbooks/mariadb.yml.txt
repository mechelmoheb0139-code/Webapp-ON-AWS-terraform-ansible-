---
- name: Configure MariaDB Database Server
  hosts: database
  become: true

  vars:
    db_name: accounts
    db_root_password: "admin123"
    db_user: admin
    db_user_password: "admin123"
    db_backup_path: /root/vprofile-project/src/main/resources/db_backup.sql

  tasks:
    - name: Update apt package cache
      ansible.builtin.apt:
        update_cache: true
        cache_valid_time: 3600

    - name: Install MariaDB and Python MySQL dependencies
      ansible.builtin.apt:
        name:
          - mariadb-server
          - python3-pymysql
        state: present

    - name: Ensure MariaDB is started and enabled
      ansible.builtin.service:
        name: mariadb
        state: started
        enabled: true

    - name: Configure MariaDB to listen on all interfaces
      ansible.builtin.lineinfile:
        path: /etc/mysql/mariadb.conf.d/50-server.cnf
        regexp: '^\s*bind-address\s*='
        line: 'bind-address = 0.0.0.0'
      notify: Restart MariaDB

    - name: Set MariaDB root password
      community.mysql.mysql_user:
        name: root
        host: localhost
        password: "{{ db_root_password }}"
        plugin: mysql_native_password
        login_user: root
        login_password: "{{ db_root_password }}"
        login_unix_socket: /run/mysqld/mysqld.sock
        state: present

    - name: Remove anonymous MariaDB users
      community.mysql.mysql_user:
        name: ""
        host_all: true
        login_user: root
        login_password: "{{ db_root_password }}"
        login_unix_socket: /run/mysqld/mysqld.sock
        state: absent

    - name: Remove test database
      community.mysql.mysql_db:
        name: test
        state: absent
        login_user: root
        login_password: "{{ db_root_password }}"
        login_unix_socket: /run/mysqld/mysqld.sock

    - name: Create accounts database
      community.mysql.mysql_db:
        name: "{{ db_name }}"
        state: present
        login_user: root
        login_password: "{{ db_root_password }}"
        login_unix_socket: /run/mysqld/mysqld.sock

    - name: Create application database user
      community.mysql.mysql_user:
        name: "{{ db_user }}"
        password: "{{ db_user_password }}"
        host: "%"
        priv: "{{ db_name }}.*:ALL"
        login_user: root
        login_password: "{{ db_root_password }}"
        login_unix_socket: /run/mysqld/mysqld.sock
        state: present


    - name: Copy database backup to MariaDB server
      ansible.builtin.copy:
        src: /root/vprofile-project/src/main/resources/db_backup.sql
        dest: /tmp/db_backup.sql
        owner: root
        group: root
        mode: '0644'

    - name: Import database backup
      community.mysql.mysql_db:
        name: "{{ db_name }}"
        state: import
        target: /tmp/db_backup.sql
        login_user: root
        login_password: "{{ db_root_password }}"
        login_unix_socket: /run/mysqld/mysqld.sock

  handlers:
    - name: Restart MariaDB
      ansible.builtin.service:
        name: mariadb
        state: restarted
